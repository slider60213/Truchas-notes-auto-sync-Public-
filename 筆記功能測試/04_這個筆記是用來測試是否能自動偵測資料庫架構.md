// 1. 定義預設排除機制與副檔名過濾
const defaultExcludes = [".git", ".obsidian", ".trash", ".smart-env", ".DS_Store"];
const ignoredExtensions = [".7z", ".mp4", ".zip", ".pdf", ".tar", ".canvas", ".h5", ".out", ".vtk"];

// 2. 嘗試讀取庫根目錄下的 .gitignore 並解析規則
let gitignoreRules = [];
const gitignoreFile = app.vault.getAbstractFileByPath(".gitignore");
if (gitignoreFile) {
    const gitignoreContent = await app.vault.read(gitignoreFile);
    gitignoreRules = gitignoreContent
        .split("\n")
        .map(line => line.trim())
        .filter(line => line && !line.startsWith("#"));
}

// 判斷路徑是否被排除
function isIgnored(path, isFolder) {
    const name = path.split("/").pop();
    
    // 基本系統資料夾排除
    if (defaultExcludes.includes(name)) return true;
    
    // 副檔名過濾
    if (!isFolder && ignoredExtensions.some(ext => name.toLowerCase().endsWith(ext))) {
        if (name === "Slider.png") return true;
        return true;
    }
    
    // .gitignore 簡易規則過濾
    for (let rule of gitignoreRules) {
        let cleanRule = rule.replace(/^\//, "").replace(/\/$/, "");
        if (cleanRule.endsWith("/*")) cleanRule = cleanRule.slice(0, -2);
        
        if (rule.startsWith("!")) continue; 
        if (path === cleanRule || path.startsWith(cleanRule + "/")) return true;
        if (name === cleanRule) return true;
    }
    
    return false;
}

// 3. 建立樹狀結構
let files = app.vault.getFiles();
let tree = {};

files.forEach(file => {
    if (isIgnored(file.path, false)) return;
    
    let parts = file.path.split("/");
    let current = tree;
    
    parts.forEach((part, index) => {
        let currentPath = parts.slice(0, index + 1).join("/");
        if (index === parts.length - 1) {
            current[part] = null;
        } else {
            if (isIgnored(currentPath, true)) return;
            if (!current[part]) current[part] = {};
            current = current[part];
        }
    });
});

// 4. 遞迴生成 Code Block 純文字樹狀圖
function generateTextTree(node, prefix = "") {
    let output = "";
    let keys = Object.keys(node).sort((a, b) => {
        let aIsDir = node[a] !== null;
        let bIsDir = node[b] !== null;
        if (aIsDir !== bIsDir) return bIsDir - aIsDir;
        return a.localeCompare(b);
    });

    keys.forEach((key, index) => {
        let isLast = index === keys.length - 1;
        let connector = isLast ? "└── " : "├── ";
        let isDir = node[key] !== null;
        let displayName = isDir ? `${key}/` : key;

        output += `${prefix}${connector}${displayName}\n`;

        if (isDir) {
            let childPrefix = prefix + (isLast ? "    " : "│   ");
            output += generateTextTree(node[key], childPrefix);
        }
    });

    return output;
}

const treeText = ".\n" + generateTextTree(tree);
