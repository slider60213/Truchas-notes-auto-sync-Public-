// 1. 定義預設排除機制（資料夾與檔案模式）
const defaultExcludes = [
    ".git", ".obsidian", ".trash", ".smart-env", ".pandoc", "tmp", "copilot", 
    "Excalidraw", "Truchas", "Zotero", "投稿", "常用梗圖"
];

// 針對副檔名或通配符的排除規則
const defaultPatternExcludes = [
    "*.7z", "*.mp4", "*.zip", "*.pdf", "*.tar", "*.canvas"
];

// 2. 嘗試讀取庫根目錄下的 .gitignore 並解析規則
let gitignoreRules = [];
const gitignoreFile = app.vault.getAbstractFileByPath(".gitignore");
if (gitignoreFile) {
    const gitignoreContent = await app.vault.read(gitignoreFile);
    gitignoreRules = gitignoreContent
        .split("\n")
        .map(line => line.trim())
        .filter(line => line && !line.startsWith("#"));
}

// 判斷資料夾或檔案路徑是否被排除
function isFolderIgnored(folderPath) {
    const parts = folderPath.split("/");
    const folderName = parts[parts.length - 1];
    
    if (!folderName) return false;

    // 基本系統/自訂排除資料夾
    if (defaultExcludes.includes(folderName)) return true;
    
    // 檢查副檔名/萬用字元排除 (例如 *.pdf, *.canvas)
    for (let pattern of defaultPatternExcludes) {
        if (pattern.startsWith("*.")) {
            const ext = pattern.slice(1); // 取得 .pdf, .canvas
            if (folderName.endsWith(ext)) return true;
        }
    }
    
    // .gitignore 簡易規則過濾
    for (let rule of gitignoreRules) {
        let cleanRule = rule.replace(/^\//, "").replace(/\/$/, "");
        
        // 處理 ! 開頭的保留例外（優先度較高）
        if (rule.startsWith("!")) {
            const allowPath = rule.slice(1).replace(/^\//, "");
            if (folderPath === allowPath || folderPath.startsWith(allowPath.replace(/\/\*\*$/, ""))) {
                return false;
            }
            continue;
        }

        if (cleanRule.endsWith("/*")) cleanRule = cleanRule.slice(0, -2);
        
        if (folderPath === cleanRule || folderPath.startsWith(cleanRule + "/")) return true;
        if (folderName === cleanRule) return true;
    }
    
    return false;
}

// 3. 取得庫中所有資料夾並建立資料夾樹狀結構
let allItems = app.vault.getAllLoadedFiles();
let folders = allItems.filter(item => item.children !== undefined && item.path !== "/");
let tree = {};

folders.forEach(folder => {
    if (isFolderIgnored(folder.path)) return;
    
    let parts = folder.path.split("/");
    let current = tree;
    
    parts.forEach((part) => {
        if (!current[part]) {
            current[part] = {};
        }
        current = current[part];
    });
});

// 4. 遞迴生成 Code Block 純文字資料夾樹狀圖（限制最多 3 層）
function generateFolderTree(node, prefix = "", depth = 1) {
    let output = "";
    let keys = Object.keys(node).sort((a, b) => a.localeCompare(b));

    keys.forEach((key, index) => {
        let isLast = index === keys.length - 1;
        let connector = isLast ? "└── " : "├── ";

        output += `${prefix}${connector}${key}/\n`;

        // 限制最多顯示 3 層 (根目錄為第 0 層，子資料夾為第 1, 2, 3 層)
        if (depth < 3) {
            let childPrefix = prefix + (isLast ? "    " : "│   ");
            output += generateFolderTree(node[key], childPrefix, depth + 1);
        }
    });

    return output;
}

const treeText = ".\n" + generateFolderTree(tree);

// 5. 在 Obsidian 視窗中渲染可直接複製的代碼區塊
dv.header(3, "📂 Repository Directory Structure (Folders Only - Max 3 Depths)");
dv.paragraph("Below is the automatically generated folder tree structure limited to 3 levels. Copy the code block directly into your `README.md`:");

dv.el("pre", "```text\n" + treeText + "```");
